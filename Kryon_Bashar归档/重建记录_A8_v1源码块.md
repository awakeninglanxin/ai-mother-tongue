/* ---------------- 几何：偶极场线 ---------------- */
function dipolePaths(layers,azimuths,L){
  const paths=[];
  for(let l=0;l<layers;l++){
    const Lk=L*(.5+.5*(l+1)/layers);
    for(let a=0;a<azimuths;a++){
      const phi=Math.PI*2*a/azimuths+(l%2)*Math.PI/azimuths;
      const pts=[],N=84;
      for(let i=0;i<=N;i++){
        const th=.13+(Math.PI-.26)*i/N, r=Lk*Math.sin(th)*Math.sin(th), s=Math.sin(th);
        pts.push(new THREE.Vector3(r*s*Math.cos(phi),r*s*Math.sin(phi),r*Math.cos(th)));
      }
      const z0=pts[N].z,z1=pts[0].z,rr=.085*Lk;
      for(let i=1;i<=18;i++){
        const s=i/18;
        pts.push(new THREE.Vector3(rr*Math.sin(Math.PI*s)*Math.cos(phi),
          rr*Math.sin(Math.PI*s)*Math.sin(phi), z0+(z1-z0)*s));
      }
      paths.push(pts);
    }
  }
  return paths;
}
function pathsToSeg(paths){

function updateThird(t){
